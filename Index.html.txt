  /** 어디가 틀렸는지 한 마디로 집어 줍니다. */
  function hintFor(t,v){
    if(!v) return '문장을 보며 입력해 주세요';
    t=t.toLowerCase(); v=v.toLowerCase();
    var i=0; while(i<t.length&&i<v.length&&t.charAt(i)===v.charAt(i)) i++;
    if(i>=t.length) return v.length>t.length?'뒤에 글자가 더 있습니다. 확인해 주세요':'문장과 다른 부분이 있습니다';
    if(i>=v.length) return t.charAt(i)===' '?'여기서 한 칸 띄어 주세요':'계속 입력해 주세요';
    if(t.charAt(i)===' ') return '한 칸 띄어 주세요';
    if(v.charAt(i)===' ') return '여기는 붙여 주세요';
    return (i+1)+'번째 글자를 확인해 주세요';
  }

        }).withFailureHandler(function(){ $('cal').style.display='none'; })
        .myStats(S.name,S.code);
    } else paintStat(r.stat);
  }
  /* 이번 달 달력을 그립니다.
     글로 '10월 중 4일' 이라고 적는 것보다 달력에 색을 칠하는 쪽이
     한눈에 읽힙니다. 어느 날 빠졌는지도 바로 보입니다. */
  function paintStat(st){
    st=st||{};
    var 날=st.dates||[], 오늘글=String(st.today||'');
    var m=/^(\d{4})-(\d{2})-(\d{2})$/.exec(오늘글);
    if(!m){ $('cal').style.display='none'; return; }
    var 해=+m[1], 달=+m[2], 오늘=+m[3];
    $('cal').style.display='';
    $('cal-t').textContent=달+'월';
    $('cal-s').textContent = 날.length ? 날.length+'일 확인' : '오늘이 첫 확인입니다';
    // 숫자로 만든 날짜라 시간대에 휘둘리지 않습니다.
    var 첫요일=new Date(해,달-1,1).getDay(), 끝날=new Date(해,달,0).getDate();
    var 한날={}; 날.forEach(function(d){ 한날[d]=1; });
    var g=$('cal-g'); g.textContent='';
    for(var i=0;i<첫요일;i++) g.appendChild(el('span','cd pad',''));
    var 초록=0;
    for(var d=1;d<=끝날;d++){
      var c='cd';
      if(한날[d]) c+=' on';
      else if(d>오늘) c+=' soon';
      if(d===오늘) c+=' now'; else if(한날[d]) 초록++;
      g.appendChild(el('span',c,d));
    }
    // 초록 칸이 하나도 없는 날(첫 확인)에는 '확인한 날' 범례가 가리킬 데가 없습니다.
    $('lg-on').style.display = 초록 ? '' : 'none';
    window.scrollTo({top:0,behavior:'smooth'});
  }

  // 시:분만 뽑습니다. 서버가 '08:07' 로 주든 '2026-10-20 08:07:00' 으로 주든 같게 나옵니다.
  function hm(v){
    var m=/(\d{1,2}):(\d{2})/.exec(String(v==null?'':v));
    return m?('0'+m[1]).slice(-2)+':'+m[2]:'';
  }
  function lock(b,on,t){ b.disabled=on; if(t) b.textContent=t; }
  function fail(id,m){ var e=$(id); e.textContent=m; e.className='err on'; }
  function hideErr(id){ $(id).className='err'; }

  /** 무사고 기록판. 날짜 계산은 서버가 끝내 놓고, 여기서는 받은 숫자만 그립니다. */
  function paintSafe(sf){
    S.safeOn=!!(sf&&sf.on);
    paintSafeBar(sf);
    if(!S.safeOn) return;
    $('sf-day').textContent=sf.day;
    var left=$('sf-left'), foot=$('sf-foot');
    left.textContent=''; foot.textContent='';
    if(sf.total){
      if(sf.done)            left.textContent='목표 달성';
      else if(sf.left===1)   left.textContent='목표까지 하루';
      else                   left.textContent='목표까지 '+sf.left+'일';
      // 막대는 그린 뒤에 늘려야 차오르는 것이 보입니다.
      setTimeout(function(){ $('sf-fill').style.width=(sf.pct||0).toFixed(1)+'%'; },80);
      foot.appendChild(el('b','',dash(sf.start)));
      foot.appendChild(document.createTextNode(' 시작 · 목표 '));
      foot.appendChild(el('b','',dash(sf.goal)));
      if(sf.goalName) foot.appendChild(document.createTextNode(' '+sf.goalName));
      foot.appendChild(document.createTextNode(' ('+sf.total+'일)'));
    } else {
      left.textContent='오늘도 이어 갑니다';
      $('sf-fill').style.width='100%';
      foot.appendChild(el('b','',dash(sf.start)));
      foot.appendChild(document.createTextNode(' 부터 이어 오고 있습니다'));
    }
  }

  /** 단계 줄에 붙는 한 줄짜리 무사고 띠. 공지 확인부터 끝까지 계속 보입니다. */
  function paintSafeBar(sf){
    var bar=$('sfline');
    if(!sf||!sf.on){ bar.className='sfline'; return; }
    bar.className='sfline on';
    $('sb-day').textContent=sf.day+'일차';
    $('sb-from').textContent=dash(sf.start)+' 시작';
    var dd=$('sb-dd');
    if(!sf.total)            dd.textContent='';
    else if(sf.left>0)       dd.textContent='D-'+sf.left;
    else if(sf.left===0)     dd.textContent='D-DAY';
    else                     dd.textContent='목표 달성';
    dd.style.display=dd.textContent?'inline-block':'none';
  }
  /** 2026-10-08 → 26.10.08 (좁은 화면에서 줄이 안 넘어가게) */
  function dash(d){
    var p=String(d||'').split('-');
    return p.length===3?(p[0].slice(2)+'.'+p[1]+'.'+p[2]):String(d||'');
  }

  function start(boot){
    $('loading').style.display='none';
    paintSafe(boot&&boot.safe); S.sf=boot&&boot.safe;
    if(!boot||!boot.ok){
      if(boot&&boot.gate&&!boot.gate.open){ shut(boot.gate); return; }
      $('loaderr-m').textContent=(boot&&boot.message)||'공지를 불러오지 못했습니다.';
      $('loaderr').style.display='flex'; return;
    }
    S.boot=boot; S.device=devId();
    if(boot.gate&&!boot.gate.open){ shut(boot.gate); return; }
    $('n-title').textContent=boot.title||'오늘 안전 공지';
    $('n-date').textContent=fmtDate(boot.date);
    if(boot.isToday===false) $('stale').style.display='flex';
    renderBody(boot.body);
    paintWhen(boot.gate);
    // 여기서부터 확인 완료 전까지는 뒤로 가기를 막습니다.
    // 중간에 빠져나가면 처음부터 다시 하셔야 하기 때문입니다.
    guard(true);
    // 매일 이름과 생년월일을 다시 칩니다. 건너뛰지 않습니다.
    // 생년월일은 매일 다시 칩니다. 이름은 이 폰에서 맞게 들어온 적이 있으면 기억해 둡니다.
    step(1); show('step-id');
    $('q-name').value=''; $('q-code').value='';
    var 기억=get('name');
    if(기억){ pickName(기억); 기억표시(); }
    else $('q-name').focus();
  }

  /** 첫 화면에 오늘 언제부터 언제까지 할 수 있는지 적어 둡니다. */
  function paintWhen(g){
    g=g||{};
    var from=g.from||'', to=g.to||'';
    var box=$('when'); box.textContent='';
    var clk=el('div','clk'), cv=el('div','cv','--:--:--');
    clk.appendChild(cv); clk.appendChild(el('div','cl','지금 시각'));
    box.appendChild(clk);
    var tx=el('div','tx'), t1=el('div','t1');
    // 닫는 시각을 따로 두지 않았다면 굳이 알리지 않습니다.
    // '자정까지' 라고 적어 두면 미뤄도 된다는 뜻으로 읽힙니다.
    if(from && to){
      t1.appendChild(document.createTextNode('오늘 '));
      t1.appendChild(el('b','',from));
      t1.appendChild(document.createTextNode(' ~ '));
      t1.appendChild(el('b','',to));
      t1.appendChild(document.createTextNode('까지'));
    } else if(from){
      t1.appendChild(document.createTextNode('오늘 '));
      t1.appendChild(el('b','',from));
      t1.appendChild(document.createTextNode(' 부터'));
    } else {
      t1.textContent='오늘 안전확인';
    }
    tx.appendChild(t1);
    tx.appendChild(el('div','t2', from && to
      ? '이 시간 안에 작성해 주세요.'
      : '출근하시면 작업 시작 전에 작성해 주세요.'));
    box.appendChild(tx);
    startClock(cv, g.nows||g.now||'');
  }

  /**
   * 서버가 알려 준 시각에서 이어서 1초씩 갑니다.
   * 기기 시계를 그냥 쓰면, 휴대폰 시각이 틀어진 사람은 열렸는데 안 열렸다고 오해합니다.
   */
  function startClock(node, serverTime){
    clearInterval(S.clock);
    var p=String(serverTime).split(':');
    var base=(+p[0]||0)*3600+(+p[1]||0)*60+(+p[2]||0);
    var at=Date.now();
    function two(n){ return (n<10?'0':'')+n; }
    function draw(){
      var t=(base+Math.floor((Date.now()-at)/1000))%86400;
      node.textContent=two(Math.floor(t/3600))+':'+two(Math.floor(t/60)%60)+':'+two(t%60);
    }
    draw(); S.clock=setInterval(draw,1000);
  }
